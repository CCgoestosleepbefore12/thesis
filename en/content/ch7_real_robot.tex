\section{OSC vs Impedance Control Equivalence}\label{sec:control_equivalence}

% Both follow: q -> J(q) -> Cartesian force -> J^T -> joint torque
%
% OSC (simulation):
%   tau = J^T * Lambda * (Kp*e + Kd*de) + nullspace + gravity
%   Lambda = (J * M^{-1} * J^T)^{-1}  (task-space inertia)
%
% Impedance (real robot):
%   tau = J^T * (Kp*e + Kd*de) + nullspace + coriolis
%   No Lambda (no inertia decoupling)
%
% Bias affects both through the SAME path: q -> J(q+b)
% Difference: Lambda introduces additional inertia compensation error in sim
% -> Real robot bias effects may differ in magnitude but not in direction
% -> Simulation is conservative (idealized)
